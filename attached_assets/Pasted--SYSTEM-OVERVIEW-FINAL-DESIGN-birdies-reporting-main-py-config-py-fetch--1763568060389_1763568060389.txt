# fetch/scanner.py
import os
from config import MAIN_FOLDER, PJR_FOLDER, REQUIRED_XML_PREFIXES

def scan_main_folder():
    files = []
    for f in os.listdir(MAIN_FOLDER):
        if any(f.upper().startswith(p) for p in REQUIRED_XML_PREFIXES):
            files.append(os.path.join(MAIN_FOLDER, f))
    return files

def scan_pjr_folder():
    cpjr_files = []
    for f in os.listdir(PJR_FOLDER):
        if f.upper().startswith("CPJR"):
            cpjr_files.append(os.path.join(PJR_FOLDER, f))
    return cpjr_files

# fetch/organizer.py
import os
import shutil
from pathlib import Path
from utils.xml_tools import extract_business_date
from config import DEST_BASE

def create_date_folder(date_str):
    year, month, day = date_str.split("-")
    base = Path(DEST_BASE) / year / month / day
    (base / "XMLFILES" / "general").mkdir(parents=True, exist_ok=True)
    (base / "JSONFILES").mkdir(parents=True, exist_ok=True)
    return base

def copy_to_date_folder(file_path, date_str):
    folder = create_date_folder(date_str)
    dest = folder / "XMLFILES" / "general" / os.path.basename(file_path)
    if not dest.exists():
        shutil.copy2(file_path, dest)
    return str(dest)

# parse/cpjr_parser.py
import xml.etree.ElementTree as ET
import pandas as pd

def parse_cpjr(cpjr_path):
    tree = ET.parse(cpjr_path)
    root = tree.getroot()

    transactions = []
    line_items = []
    loyalty_items = []

    for sale in root.findall(".//SaleEvent"):
        tx_id = sale.findtext("TransactionID", "")
        cashier = sale.findtext("CashierID", "")
        date = sale.findtext("EventStartDate", "")
        time = sale.findtext("EventStartTime", "")
        full_dt = f"{date} {time}"

        fuel_vol = 0
        fuel_amt = 0
        merch_amt = 0
        tender_type = sale.findtext(".//Tender/TenderCode", "")

        # Fuel lines
        for fl in sale.findall(".//FuelLine"):
            qty = float(fl.findtext("SalesQuantity", "0"))
            amt = float(fl.findtext("SalesAmount", "0"))
            desc = fl.findtext("Description", "")
            pump = fl.findtext("FuelPositionID", "")

            fuel_vol += qty
            fuel_amt += amt

            line_items.append({
                "transaction_id": tx_id,
                "type": "fuel",
                "description": desc,
                "pump": pump,
                "quantity": qty,
                "amount": amt,
                "tender": tender_type,
                "datetime": full_dt
            })

            # Loyalty
            promo = fl.find("Promotion")
            if promo is not None:
                loyalty_items.append({
                    "transaction_id": tx_id,
                    "promotion_id": promo.findtext("PromotionID", ""),
                    "promotion_amount": float(promo.findtext("PromotionAmount", "0")),
                    "datetime": full_dt
                })

        # Item lines
        for il in sale.findall(".//ItemLine"):
            upc = il.findtext("./ItemCode/POSCode", "")
            desc = il.findtext("Description", "")
            qty = float(il.findtext("SalesQuantity", "0"))
            amt = float(il.findtext("SalesAmount", "0"))

            merch_amt += amt

            line_items.append({
                "transaction_id": tx_id,
                "type": "merch",
                "upc": upc,
                "description": desc,
                "quantity": qty,
                "amount": amt,
                "tender": tender_type,
                "datetime": full_dt
            })

        transactions.append({
            "transaction_id": tx_id,
            "datetime": full_dt,
            "cashier": cashier,
            "fuel_volume": fuel_vol,
            "fuel_amount": fuel_amt,
            "merch_amount": merch_amt,
            "total_amount": fuel_amt + merch_amt
        })

    return (
        pd.DataFrame(transactions),
        pd.DataFrame(line_items),
        pd.DataFrame(loyalty_items)
    )

# parse/fgm_parser.py
import xml.etree.ElementTree as ET
import pandas as pd

def parse_fgm(path):
    tree = ET.parse(path)
    root = tree.getroot()

    grades = []

    for d in root.findall(".//FGMDetail"):
        grade = d.findtext("FuelGradeID", "")
        vol = float(d.findtext(".//FuelGradeSalesVolume", "0"))
        amt = float(d.findtext(".//FuelGradeSalesAmount", "0"))
        disc = float(d.findtext(".//DispenserDiscountAmount", "0"))

        grades.append({
            "grade": grade,
            "volume": vol,
            "amount": amt,
            "discount_amount": disc
        })

    return pd.DataFrame(grades)

# parse/mcm_parser.py
import xml.etree.ElementTree as ET
import pandas as pd

def parse_mcm(path):
    tree = ET.parse(path)
    root = tree.getroot()

    rows = []
    for d in root.findall(".//MCMDetail"):
        code = d.findtext("MerchandiseCode", "")
        desc = d.findtext("MerchandiseCodeDescription", "")
        amt = float(d.findtext(".//SalesAmount", "0"))
        qty = float(d.findtext(".//SalesQuantity", "0"))
        cnt = float(d.findtext(".//TransactionCount", "0"))

        rows.append({
            "department_code": code,
            "description": desc,
            "sales_amount": amt,
            "quantity": qty,
            "transaction_count": cnt
        })

    return pd.DataFrame(rows)

# main.py
from fetch.scanner import scan_main_folder, scan_pjr_folder
from fetch.organizer import copy_to_date_folder
from fetch.journal_matcher import get_cpjr_date

from parse.cpjr_parser import parse_cpjr
from parse.fgm_parser import parse_fgm
from parse.mcm_parser import parse_mcm
from parse.ism_parser import parse_ism

from transform.fuel_reports import build_fuel_summary
from export.generate_json import save_json

import os
from pathlib import Path

def process_day(date_str, xml_folder, json_folder):
    # Find required files
    cpjr = next((str(p) for p in Path(xml_folder).iterdir() if p.name.startswith("CPJR")), None)
    fgm  = next((str(p) for p in Path(xml_folder).iterdir() if p.name.startswith("FGM")), None)
    mcm  = next((str(p) for p in Path(xml_folder).iterdir() if p.name.startswith("MCM")), None)
    ism  = next((str(p) for p in Path(xml_folder).iterdir() if p.name.startswith("ISM")), None)

    # Parse
    tx_df, line_df, loyalty_df = parse_cpjr(cpjr)
    fgm_df = parse_fgm(fgm)
    mcm_df = parse_mcm(mcm)
    ism_df = parse_ism(ism)

    # Transform
    fuel_summary = build_fuel_summary(fgm_df)

    # Export
    save_json(fuel_summary, Path(json_folder) / "fuel_summary.json")

def main():
    # 1. Scan folders
    main_files = scan_main_folder()
    cpjr_files = scan_pjr_folder()

    # 2. Organize
    for f in main_files:
        date = get_cpjr_date(f) or "2025-01-01"
        copy_to_date_folder(f, date)

    for f in cpjr_files:
        date = get_cpjr_date(f)
        copy_to_date_folder(f, date)

    # 3. Process organized folders
    for year in Path(DEST_BASE).iterdir():
        for month in year.iterdir():
            for day in month.iterdir():
                xml_folder = day / "XMLFILES" / "general"
                json_folder = day / "JSONFILES"
                process_day(day.name, xml_folder, json_folder)

if __name__ == "__main__":
    main()
