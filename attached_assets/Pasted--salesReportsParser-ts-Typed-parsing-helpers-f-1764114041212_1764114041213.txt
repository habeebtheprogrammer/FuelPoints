// salesReportsParser.ts
// ---------------------------------------------
// Typed parsing helpers for Sales Analytics JSON
// ---------------------------------------------
//
// This module takes the raw JSON returned from your
// REST endpoints and converts it into strongly-typed
// structures that are easy to bind to tables / metric
// cards in a front-end UI.

// ---------- Shared Types ----------

export type ReportId =
  | 'dailyTotalSales'
  | 'dailyFuelSales'
  | 'fuelByGrade'
  | 'departmentSales'
  | 'aggregatedItems'
  | 'unknownItems'
  | 'storeSummary'
  | 'fuelDispensers'
  | 'transactionsOverview'
  | 'transactionLineItems'
  | 'loyaltyDetails'
  | 'loyaltyOverview';

export type DataType = 'string' | 'number' | 'currency' | 'date' | 'datetime';

export interface ColumnDef<K extends string = string> {
  /** Stable key used by the row objects */
  id: K;
  /** Human‑readable column title */
  label: string;
  /** Hint for rendering / formatting */
  dataType?: DataType;
}

export interface TableReport<
  Row extends Record<string, any> = Record<string, any>,
  K extends keyof Row & string = keyof Row & string
> {
  kind: 'table';
  reportId: ReportId;
  columns: ColumnDef<K>[];
  rows: Row[];
}

export interface MetricCard {
  id: string;
  label: string;
  value: number;
  dataType?: Exclude<DataType, 'date' | 'datetime'>;
}

export interface MetricsReport {
  kind: 'metrics';
  reportId: ReportId;
  cards: MetricCard[];
}

export type ParsedReport = TableReport<any> | MetricsReport;

// ---------- Internal Helpers ----------

type AnyRecord = Record<string, any>;
type NumericLike = string | number | null | undefined;

/**
 * Normalize the common API response shapes to an array of records.
 * You can adjust this if your API uses a different envelope.
 */
function asArray(raw: unknown): AnyRecord[] {
  if (Array.isArray(raw)) {
    return raw as AnyRecord[];
  }

  if (raw && typeof raw === 'object') {
    const container = raw as any;
    const candidate =
      container.data ??
      container.items ??
      container.rows ??
      container.result ??
      container.results;

    if (Array.isArray(candidate)) {
      return candidate as AnyRecord[];
    }
  }

  throw new Error(
    'Expected an array of rows or an object with a data/items/rows/result/results array'
  );
}

/** Best‑effort numeric coercion that accepts numbers or numeric strings (optionally with $ or commas). */
function toNumber(value: NumericLike): number {
  if (value === null || value === undefined || value === '') {
    return 0;
  }
  if (typeof value === 'number') {
    return value;
  }
  const cleaned = String(value).replace(/[^0-9.\-]/g, '');
  const n = parseFloat(cleaned);
  return Number.isNaN(n) ? 0 : n;
}

function getField(obj: AnyRecord, ...keys: string[]): unknown {
  for (const key of keys) {
    if (key in obj) {
      return (obj as any)[key];
    }
  }
  return undefined;
}

function getStringField(obj: AnyRecord, ...keys: string[]): string {
  const raw = getField(obj, ...keys);
  if (raw === null || raw === undefined) {
    return '';
  }
  return String(raw);
}

function getNumberField(obj: AnyRecord, ...keys: string[]): number {
  const raw = getField(obj, ...keys) as NumericLike;
  return toNumber(raw);
}

/** For metric‑style endpoints that may return either an object or a single‑row array. */
function asObject(raw: unknown): AnyRecord {
  if (raw && typeof raw === 'object' && !Array.isArray(raw)) {
    return raw as AnyRecord;
  }
  if (Array.isArray(raw) && raw.length > 0) {
    const first = raw[0];
    if (first && typeof first === 'object') {
      return first as AnyRecord;
    }
  }
  return {};
}

// ---------- Per‑Endpoint Public Parsers ----------

export function parseDailyTotalSales(
  raw: unknown
): TableReport<DailyTotalSalesRow> {
  const rows = asArray(raw).map(mapDailyTotalSalesRow);
  return {
    kind: 'table',
    reportId: 'dailyTotalSales',
    columns: dailyTotalSalesColumns,
    rows,
  };
}

export function parseDailyFuelSales(

    kind: 'table',
    reportId: 'dailyFuelSales',
    columns: dailyFuelSalesColumns,
    rows,
  };
}

export function parseFuelByGrade(
  raw: unknown
): TableReport<FuelByGradeRow> {
  const rows = asArray(raw).map(mapFuelByGradeRow);

  // Include Date column only when at least one row has a non‑empty date
  const includeDate = rows.some((r) => !!r.date);

  const dateColumn: ColumnDef<'date'> = {
    id: 'date',
    label: 'Date',
    dataType: 'date',
  };

  const columns: ColumnDef<string>[] = includeDate
    ? [dateColumn, ...fuelByGradeBaseColumns]
    : [...fuelByGradeBaseColumns];

  return {
    kind: 'table',
    reportId: 'fuelByGrade',
    columns: columns as ColumnDef<keyof FuelByGradeRow & string>[],
    rows,
  };
}

// ---------- Dispatcher Helper ----------

/**
 * Convenience function that picks the correct parser based on endpoint path.
 *
 * Example usage:
 *
 *   const res = await fetch('/api/sales/reports/daily-total-sales');
 *   const json = await res.json();
 *   const parsed = parseReportFromEndpoint(res.url, json);
 */
export function parseReportFromEndpoint(
  endpoint: string,
  raw: unknown
): ParsedReport {
  // Strip query string and host so we can match on the path only
  const path = endpoint.split('?')[0];

  if (path.endsWith('/daily-total-sales')) {
    return parseDailyTotalSales(raw);
  }
  if (path.endsWith('/daily-fuel-sales')) {
    return parseDailyFuelSales(raw);
  }
  if (path.endsWith('/fuel-by-grade')) {
    return parseFuelByGrade(raw);
  }
  if (path.endsWith('/department-sales')) {
    return parseDepartmentSales(raw);
  }
  if (path.endsWith('/aggregated-items')) {
    return parseAggregatedItems(raw);
  }
  if (path.endsWith('/unknown-items')) {
    return parseUnknownItems(raw);
  }
  if (path.endsWith('/store-summary')) {
    return parseStoreSummary(raw);
  }
  if (path.endsWith('/fuel-dispensers')) {
    return parseFuelDispensers(raw);
  }
  if (path.endsWith('/transactions')) {
    return parseTransactionsOverview(raw);
  }
  if (path.endsWith('/transaction-line-items')) {
    return parseTransactionLineItems(raw);
  }
  if (path.endsWith('/loyalty-details')) {
    return parseLoyaltyDetails(raw);
  }
  if (path.endsWith('/loyalty-overview')) {
    return parseLoyaltyOverview(raw);
  }

  throw new Error(`No parser configured for endpoint: ${endpoint}`);
}
